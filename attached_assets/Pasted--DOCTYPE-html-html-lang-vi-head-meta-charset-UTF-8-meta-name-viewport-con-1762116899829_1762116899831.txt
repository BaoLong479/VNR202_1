    <script>
        function initCharts() {
            if (typeof Chart === 'undefined') {
                console.error('Chart.js failed to load. Cannot create charts.');
                return;
            }

            console.log('Chart.js is loaded, creating charts...');

            const chartTooltipCallback = {
                plugins: {
                    tooltip: {
                        callbacks: {
                            title: function(tooltipItems) {
                                const item = tooltipItems[0];
                                let label = item.chart.data.labels[item.dataIndex];
                                if (Array.isArray(label)) {
                                    return label.join(' ');
                                } else {
                                    return label;
                                }
                            }
                        }
                    }
                },
                responsive: true,
                maintainAspectRatio: false
            };

            const inflationCtx = document.getElementById('inflationChart').getContext('2d');
            new Chart(inflationCtx, {
                type: 'line',
                data: {
                    labels: ['1980', '1981', '1982', '1983', '1984', '1985', '1986'],
                    datasets: [{
                        label: 'Tỷ lệ lạm phát (%)',
                        data: [25.1, 69.1, 95.4, 49.3, 100.5, 385.0, 774.7],
                        backgroundColor: 'rgba(231, 76, 60, 0.1)',
                        borderColor: '#E74C3C',
                        borderWidth: 3,
                        pointBackgroundColor: '#E74C3C',
                        tension: 0.1
                    }]
                },
                options: chartTooltipCallback
            });

            const gdpCtx = document.getElementById('gdpChart').getContext('2d');
            new Chart(gdpCtx, {
                type: 'bar',
                data: {
                    labels: ['TB 1976-1980', '1981', '1982', '1983', '1984', '1985'],
                    datasets: [{
                        label: 'Tăng trưởng GDP (%)',
                        data: [0.4, 5.1, 8.8, 7.7, 8.3, 5.3],
                        backgroundColor: '#3498DB',
                        borderColor: '#2980B9',
                        borderWidth: 1
                    }]
                },
                options: chartTooltipCallback
            });

            const foodCtx = document.getElementById('foodChart').getContext('2d');
            const foodLabels = [['Sản lượng TB', '1976-1980'], ['Sản lượng TB', '1981-1985']];
            new Chart(foodCtx, {
                type: 'bar',
                data: {
                    labels: foodLabels,
                    datasets: [{
                        label: 'Sản lượng lương thực (Triệu tấn)',
                        data: [13.4, 17.0],
                        backgroundColor: ['#F39C12', '#F1C40F'],
                        borderColor: ['#E67E22', '#D4AC0D'],
                        borderWidth: 1
                    }]
                },
                options: chartTooltipCallback
            });

            const industryCtx = document.getElementById('industryChart').getContext('2d');
            const industryLabels = ['Hoạt động hiệu quả', ['Hoạt động cầm chừng', '(dưới 50% công suất)']];
            new Chart(industryCtx, {
                type: 'doughnut',
                data: {
                    labels: industryLabels,
                    datasets: [{
                        data: [50, 50],
                        backgroundColor: ['#3498DB', '#E74C3C'],
                        hoverOffset: 4
                    }]
                },
                options: chartTooltipCallback
            });
        }

        window.onload = function() {
            initCharts();
        };
    </script>
